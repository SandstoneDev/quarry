# CLEO-like runnable routines, spawned on demand from the debug menu "Scripts" tab.
# MAIN terminates immediately; each `e_*` label is an exported entry point
# (scm_asm.py --syms emits SCMSYM_<NAME>). Routines are linear + self-terminating.
GLOBALS 6
# world init from the script (as SA main.scm does): set the start time of day.
# The hardcoded CClock default still runs first; this overrides it -> time is now
# script-configured. Add weather/etc. here to migrate more world init.

  SET_TIME_OF_DAY 8 0
  RELEASE_WEATHER
  TERMINATE
:e_midday
  SET_TIME_OF_DAY 12 0
  TERMINATE
:e_night

  TERMINATE

# resident proximity-trigger (mission primitive): loops until the player reaches
# the point, then rewards + prints. Start it, walk to (2490,-1670), it fires.
:e_trigger
  ADD_BLIP_FOR_COORD 2490.0 -1670.0 13.0 $0
  CREATE_OBJECT 2490.0 -1670.0 13.0 $1
:t_loop
  WAIT 250
  IS_PLAYER_NEAR_2D 2490.0 -1670.0 15.0
  GOTO_IF_FALSE @t_loop
  REMOVE_BLIP $0
  DELETE_OBJECT $1
  PRINT_STRING "DONE" 3000
  ADD_PLAYER_MONEY 250
  TERMINATE

# --- TEST MISSION: reach the checkpoint -> Mission Passed -----------------
# Authored with the SA-canonical opcodes (not the demake 0x7Fxx set) so this IS the
# shape real main.scm missions have - port them 1:1 by pasting their SANNY listing:
#   DECLARE_MISSION_FLAG $ONMISSION ; $ONMISSION = 1 ; ...objective... ; blip + marker ;
#   loop: LOCATE_PLAYER ; on arrival REGISTER_MISSION_PASSED + tune ; $ONMISSION = 0 ; end
# $0 = ON_MISSION flag, $1 = blip handle, $2 = marker object handle.
:e_mtest
  DECLARE_MISSION_FLAG $0
  SET_VAR_INT $0 1
  PRINT_HELP "MTEST_H"
  ADD_BLIP_FOR_COORD 2490.0 -1655.0 13.0 $1
:m_loop
  WAIT 250
  LOCATE_PLAYER_ANY_MEANS_2D 0 2490.0 -1655.0 4.0 4.0 0
  GOTO_IF_FALSE @m_loop
  REMOVE_BLIP $1
  CLEAR_HELP
  REGISTER_MISSION_PASSED "MTEST_N"
  PLAY_MISSION_PASSED_TUNE 1
  SET_VAR_INT $0 0
  TERMINATE
